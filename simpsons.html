Character Co-occurrence

shared_scenes <- simpsons_script_lines |> group_by(episode_id, location_id) |> select(raw_character_text) |> drop_na(raw_character_text) |> unique() |> nest(data = raw_character_text) 
Adding missing grouping variables: `episode_id`, `location_id`
character_cooc <- shared_scenes |> ungroup() |> select(data) |> pull(1) |> map(\(x){
      if(nrow(x) > 1){
           x |> pull(1) |> combn(2) |> t() |> as_tibble()
      } 
}) |> list_rbind() 
character_cooc_net <- character_cooc |> group_by(V1, V2) |>
    summarise(weight = n(), .groups = "keep") |> arrange(desc(weight))
simpsons_graph <- character_cooc_net |> filter(weight > 10) |>
    graph_from_data_frame(directed = F) 

simpsons_graph|> 
        ggraph(layout = "stress") +
        geom_edge_link(width = 1, colour = "orange") +
