# Merge the two files on the country code and year
democracypop = pd.merge(democracies, populations, left_on=['country_code', 'year'], 
                        right_on=['Country Code', 'Year'], how='inner')

popsum = (democracypop.groupby(['year','is_democracy'])['Value'].sum()).to_frame().reset_index()

# Create the graph

plt.figure(figsize=(16, 8))
plt.style.use('dark_background')
custom_palette = {False: 'teal', True: 'lime'}
lineplt = sns.lineplot(data=popsum, x='year', y='Value', hue='is_democracy', 
                       palette=custom_palette, linewidth = 4, hue_order=[True, False])

lineplt.set_title("Global Population by Democracy Status", fontsize=30)
lineplt.set_ylabel("Population (in billions)", fontsize=18)
lineplt.set_xlabel("Year", fontsize=18)
lineplt.axvline(1998, color="#666")
lineplt.axvline(1984, color="#666")
lineplt.axvline(2007, color="#666")
plt.text(1998, popsum['Value'].max()/4, '1998: Indonesia and \nNigeria switch', rotation=45, verticalalignment='bottom', color='#666', fontsize=18)
plt.text(1984, popsum['Value'].max()/4, '1984: Brazil and \nUruguay switch', rotation=45, verticalalignment='bottom', color='#666', fontsize=18)
plt.text(2007, popsum['Value'].max()/4,' 2007: Thailand alternates, \nPakistan switches',  rotation=45, verticalalignment='bottom', color='#666', fontsize=18)

plt.xticks(fontsize=14)
plt.yticks(fontsize=14)

plt.legend(title='Democratic Status', fontsize = 18, title_fontsize = 14)

plt.figtext(0.99, 0.04, '@paulusm@scholar.social #TidyTuesday', 
            horizontalalignment='right', fontsize=14, alpha=0.7, fontstyle='italic')
plt.figtext(0.99, 0.01, 'Data : Bjørnskov & Rode (2020; democracies); World Bank (populations)', 
            horizontalalignment='right', fontsize=14, alpha=0.7 , fontstyle='italic')

plt.show()
